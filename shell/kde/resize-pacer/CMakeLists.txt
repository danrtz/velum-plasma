cmake_minimum_required(VERSION 3.20)
project(velum_resize_pacer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(BUILD_SHARED_LIBS ON)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins")
set(QT_MAJOR_VERSION 6)
set(CMAKE_AUTOMOC ON)
find_package(ECM REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs6)
find_package(Qt6 REQUIRED COMPONENTS Core CorePrivate Widgets)
find_package(KWin REQUIRED)
find_package(KF6CoreAddons REQUIRED)
kcoreaddons_add_plugin(velum_resize_pacer SOURCES pacer.cpp INSTALL_NAMESPACE "kwin/effects/plugins")
target_link_libraries(velum_resize_pacer PRIVATE KWin::kwin Qt6::Core Qt6::CorePrivate)
set_target_properties(velum_resize_pacer PROPERTIES PREFIX "")
# A new filename lets the live compositor load this revision without restarting.
set_target_properties(velum_resize_pacer PROPERTIES OUTPUT_NAME "velum_resize_pacer_v2")
